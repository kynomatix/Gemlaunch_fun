python# Don't use gql library - use raw requests with GraphQL
import requests
import json

BLOCKSCOUT_GRAPHQL = "https://explorer.testnet.kasplextest.xyz/graphiql"

def graphql_query(query, variables=None):
    """Execute GraphQL query using raw requests"""
    payload = {
        "query": query,
        "variables": variables or {}
    }
    
    response = requests.post(
        BLOCKSCOUT_GRAPHQL,
        json=payload,
        headers={
            "Content-Type": "application/json",
            "Accept": "application/json"
        }
    )
    
    # Check if response is JSON
    if 'application/json' in response.headers.get('Content-Type', ''):
        return response.json()
    else:
        print(f"Got HTML instead of JSON. Status: {response.status_code}")
        print(f"Response preview: {response.text[:500]}")
        return None
