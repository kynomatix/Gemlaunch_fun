#### **Issue #3: Smart Contract Should Pre-Validate**

Your current contract will fail midway through the loop if allowance runs out:

```solidity
// Current: Fails on transfer #15 if allowance insufficient
for (uint256 i = 0; i < recipients.length; i++) {
    transferFrom(...);  // ← Could fail here, wasting gas
}
```

**Fix:** Check total upfront:

```solidity
function batchTransfer(...) external nonReentrant {
    // Calculate total first
    uint256 totalNeeded = 0;
    for (uint256 i = 0; i < amounts.length; i++) {
        totalNeeded += amounts[i];
    }
    
    // Pre-validate allowance
    require(
        tokenContract.allowance(msg.sender, address(this)) >= totalNeeded,
        "Insufficient allowance"
    );
    
    // Pre-validate balance
    require(
        tokenContract.balanceOf(msg.sender) >= totalNeeded,
        "Insufficient balance"
    );
    
    // Now do transfers
    for (uint256 i = 0; i < recipients.length; i++) {
        transferFrom(...);
    }
}
```

This way it fails **fast** (before wasting gas on partial transfers) instead of failing on recipient #15.

### **Smart Contract**

```solidity
contract AirdropDistributor is ReentrancyGuard {
    event AirdropDistributed(
        address indexed token,
        address indexed distributor,
        uint256 recipientCount,
        uint256 totalAmount
    );
    
    function batchTransfer(
        address token,
        address[] calldata recipients,
        uint256[] calldata amounts
    ) external nonReentrant {
        require(recipients.length == amounts.length, "Length mismatch");
        require(recipients.length > 0, "Empty");
        
        IERC20 tokenContract = IERC20(token);
        
        // Calculate total and pre-validate
        uint256 total = 0;
        for (uint256 i = 0; i < amounts.length; i++) {
            require(amounts[i] > 0, "Invalid amount");
            total += amounts[i];
        }
        
        require(tokenContract.allowance(msg.sender, address(this)) >= total, "Insufficient allowance");
        require(tokenContract.balanceOf(msg.sender) >= total, "Insufficient balance");
        
        // Execute transfers
        for (uint256 i = 0; i < recipients.length; i++) {
            require(recipients[i] != address(0), "Invalid recipient");
            require(tokenContract.transferFrom(msg.sender, recipients[i], amounts[i]), "Transfer failed");
        }
        
        emit AirdropDistributed(token, msg.sender, recipients.length, total);
    }
}
```

That's it. Simple, secure, does the job.
