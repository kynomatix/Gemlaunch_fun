pythondef _predict_create2_address(self, salt: bytes, init_code_hash: bytes) -> str:
    """
    Predict CREATE2 contract address
    address = keccak256(0xff ++ factory_address ++ salt ++ keccak256(init_code))[12:]
    """
    from eth_utils import keccak, to_checksum_address
    
    # 0xff prefix
    prefix = bytes.fromhex('ff')
    
    # Factory address (20 bytes)
    factory_address = bytes.fromhex(self.token_factory_address[2:])
    
    # Concatenate: 0xff ++ factory ++ salt ++ init_code_hash
    data = prefix + factory_address + salt + init_code_hash
    
    # Hash and take last 20 bytes
    hash_result = keccak(data)
    contract_address = hash_result[-20:]
    
    return to_checksum_address('0x' + contract_address.hex())

def deploy_token_via_factory(self, name, symbol, total_supply, creator, **kwargs):
    """
    Oracle wallet deploys token via TokenFactory using CREATE2
    """
    # Generate salt from creator address + timestamp
    import time
    salt_input = creator + str(int(time.time()))
    salt = Web3.keccak(text=salt_input)
    
    # Calculate init code hash (from TokenFactory contract)
    # This should match the hash in your TokenFactory.sol
    # You need to get this from your deployed factory contract
    init_code_hash = self.token_factory_contract.functions.POOL_INIT_CODE_HASH().call()
    
    # Predict contract address using CREATE2
    contract_address = self._predict_create2_address(salt, init_code_hash)
    
    # Build transaction with salt
    tx = self.token_factory_contract.functions.createToken(
        name=name,
        symbol=symbol,
        totalSupply=Web3.to_wei(total_supply, 'ether'),
        creator=Web3.to_checksum_address(creator),
        reservedPercentage=kwargs.get('reserved_percentage', 0),
        antiBotEnabled=kwargs.get('anti_bot_enabled', False),
        salt=salt  # ← Pass salt for CREATE2
    ).build_transaction({
        'from': self.oracle_account.address,
        'nonce': self.w3.eth.get_transaction_count(self.oracle_account.address),
        'gas': 2000000,
        'gasPrice': self.w3.eth.gas_price
    })
    
    return {
        'tx_data': tx,
        'contract_address': contract_address,
        'salt': salt.hex()  # Store for verification
    }

pythondef verify_deployment(self, tx_hash: str, predicted_address: str) -> bool:
    """
    Verify that the contract was actually deployed to the predicted address
    """
    receipt = self.w3.eth.get_transaction_receipt(tx_hash)
    
    # Find TokenCreated event in logs
    for log in receipt['logs']:
        try:
            # Decode event (assuming first topic is event signature)
            if log['topics'][0].hex() == self._get_token_created_signature():
                # Extract contract address from event
                actual_address = '0x' + log['topics'][1].hex()[-40:]  # Last 20 bytes
                return actual_address.lower() == predicted_address.lower()
        except:
            continue
    
    return False

def _get_token_created_signature(self) -> str:
    """
    Get TokenCreated event signature hash
    event TokenCreated(address indexed tokenAddress, address indexed creator, ...)
    """
    from eth_utils import keccak
    signature = "TokenCreated(address,address,string,string,uint256)"
    return '0x' + keccak(text=signature).hex()

python@app.route('/api/relay/transaction', methods=['POST'])
@csrf.exempt
def relay_signed_transaction():
    """
    Relay a signed transaction to the blockchain
    Used for wallets that sign but don't broadcast (Kastle/KasWare)
    """
    data = request.get_json()
    
    try:
        signed_tx = data.get('signed_tx')
        if not signed_tx:
            return jsonify({'success': False, 'error': 'Missing signed_tx'}), 400
        
        # Broadcast to blockchain
        tx_hash = web3_service.w3.eth.send_raw_transaction(signed_tx)
        
        return jsonify({
            'success': True,
            'tx_hash': tx_hash.hex()
        })
        
    except Exception as e:
        app.logger.error(f"Relay transaction failed: {e}")
        return jsonify({'success': False, 'error': str(e)}), 500

pythonfrom flask import Response, stream_with_context
import json
import time

@app.route('/api/tx/<tx_hash>/stream')
def stream_transaction_status(tx_hash):
    """
    Server-Sent Events stream for transaction status
    Polls blockchain every 2 seconds and streams updates
    """
    def generate():
        max_polls = 150  # 5 minutes (150 * 2 seconds)
        poll_count = 0
        
        while poll_count < max_polls:
            try:
                # Check transaction status
                receipt = web3_service.w3.eth.get_transaction_receipt(tx_hash)
                
                if receipt:
                    # Transaction confirmed
                    status = 'confirmed' if receipt['status'] == 1 else 'failed'
                    
                    yield f"data: {json.dumps({
                        'status': status,
                        'tx_hash': tx_hash,
                        'block_number': receipt['blockNumber'],
                        'gas_used': receipt['gasUsed'],
                        'logs': [{'address': log['address'], 'topics': [t.hex() for t in log['topics']]} for log in receipt['logs']]
                    })}\n\n"
                    
                    break  # Done
                else:
                    # Still pending
                    yield f"data: {json.dumps({
                        'status': 'pending',
                        'tx_hash': tx_hash,
                        'poll_count': poll_count
                    })}\n\n"
                
            except Exception as e:
                yield f"data: {json.dumps({
                    'status': 'error',
                    'error': str(e)
                })}\n\n"
                break
            
            time.sleep(2)
            poll_count += 1
        
        # Timeout
        if poll_count >= max_polls:
            yield f"data: {json.dumps({
                'status': 'timeout',
                'message': 'Transaction confirmation timed out'
            })}\n\n"
    
    return Response(
        stream_with_context(generate()),
        mimetype='text/event-stream',
        headers={
            'Cache-Control': 'no-cache',
            'X-Accel-Buffering': 'no'
        }
    )
