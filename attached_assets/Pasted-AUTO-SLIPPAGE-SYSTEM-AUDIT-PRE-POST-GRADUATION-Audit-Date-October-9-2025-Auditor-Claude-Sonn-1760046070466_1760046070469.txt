function _getPriceVolatility(address poolAddress) internal view returns (uint256) {
    IUniswapV3Pool pool = IUniswapV3Pool(poolAddress);
    
    // Get current price
    (uint160 sqrtPriceX96, , , , , , ) = pool.slot0();
    uint256 currentPrice = _sqrtPriceToPrice(sqrtPriceX96);
    
    // Get 5-minute TWAP price
    uint32[] memory secondsAgos = new uint32[](2);
    secondsAgos[0] = 300; // 5 minutes ago
    secondsAgos[1] = 0;   // now
    
    (int56[] memory tickCumulatives, ) = pool.observe(secondsAgos);
    int24 avgTick = int24((tickCumulatives[1] - tickCumulatives[0]) / 300);
    uint256 twapPrice = _tickToPrice(avgTick);
    
    // Calculate volatility (price deviation)
    uint256 deviation = currentPrice > twapPrice
        ? ((currentPrice - twapPrice) * 10000) / twapPrice
        : ((twapPrice - currentPrice) * 10000) / twapPrice;
    
    return deviation; // Returns basis points (e.g., 300 = 3%)
}

        'slippage_bps': total_slippage,
        'slippage_percent': total_slippage / 100,
        'risk_level': risk_level,
        'breakdown': {
            'base': base_slippage,
            'volatility': volatility_adjustment,
            'liquidity': liquidity_adjustment
        },
        'pool_stats': {
            'volatility_bps': volatility,
            'liquidity_kas': Web3.from_wei(liquidity_kas, 'ether'),
            'trade_impact_bps': trade_impact_bps
        }
    }

async def get_quote_with_slippage(
    self,
    token_in: str,
    token_out: str,
    amount_in: int,
    pool_address: str
) -> dict:
    """
    Get DEX quote and calculate minAmountOut with auto-slippage
    """
    
    # Get quote from QuoterV2
    quote_params = {
        'tokenIn': token_in,
        'tokenOut': token_out,
        'amountIn': amount_in,
        'fee': 2500,  # 0.25% fee tier
        'sqrtPriceLimitX96': 0
    }
    
    try:
        result = await self.quoter.functions.quoteExactInputSingle(
            quote_params
        ).call()
        
        amount_out = result[0]  # Expected output
        
    except Exception as e:
        raise Exception(f"QuoterV2 call failed: {e}")
    
    # Calculate optimal slippage
    slippage_data = await self.calculate_dex_slippage(
        pool_address, token_in, token_out, amount_in
    )
    
    # Apply slippage
    min_amount_out = (amount_out * (10000 - slippage_data['slippage_bps'])) // 10000
    
    return {
        'amount_out': amount_out,
        'min_amount_out': min_amount_out,
        'slippage_bps': slippage_data['slippage_bps'],
        'risk_level': slippage_data['risk_level'],
        'pool_stats': slippage_data['pool_stats']
    }

async def execute_swap_with_retry(
    self,
    token_in: str,
    token_out: str,
    amount_in: int,
    recipient: str,
    pool_address: str,
    max_retries: int = 3
) -> dict:
    """
    Execute DEX swap with intelligent retry on slippage failure
    """
    
    for attempt in range(1, max_retries + 1):
        try:
            # Get quote with auto-slippage
            quote = await self.get_quote_with_slippage(
                token_in, token_out, amount_in, pool_address
            )
            
            # Build swap transaction
            swap_tx = await self._build_swap_transaction(
                token_in=token_in,
                token_out=token_out,
                amount_in=amount_in,
                min_amount_out=quote['min_amount_out'],
                recipient=recipient
            )
            
            # Send transaction
            tx_hash = await self.w3.eth.send_transaction(swap_tx)
            receipt = await self.w3.eth.wait_for_transaction_receipt(tx_hash)
            
            if receipt.status == 1:
                return {
                    'success': True,
                    'tx_hash': tx_hash.hex(),
                    'amount_out': quote['amount_out'],
                    'min_amount_out': quote['min_amount_out'],
                    'slippage_used': quote['slippage_bps'],
                    'attempt': attempt
                }
        
        except Exception as e:
            error_msg = str(e).lower()
            
            # Check if it's a slippage error
            if ('slippage' in error_msg or 'too little received' in error_msg) and attempt < max_retries:
                # Calculate increased slippage for retry
                new_slippage = quote['slippage_bps'] + (100 * attempt)  # +1% per attempt
                if new_slippage > 1500:
                    new_slippage = 1500
                
                print(f"Retry {attempt}: Increasing slippage to {new_slippage/100}%")
                continue
            else:
                raise Exception(f"Swap failed after {attempt} attempts: {e}")
    
    raise Exception("Max retries exceeded")

async def _get_price_volatility(self, pool_address: str) -> int:
    """
    Calculate price volatility using TWAP oracle
    Returns: volatility in basis points
    """
    try:
        pool_contract = self.w3.eth.contract(
            address=pool_address,
            abi=UNISWAP_V3_POOL_ABI
        )
        
        # Get current price
        slot0 = await pool_contract.functions.slot0().call()
        sqrt_price_x96 = slot0[0]
        current_price = self._sqrt_price_to_price(sqrt_price_x96)
        
        # Get 5-minute TWAP
        seconds_agos = [300, 0]  # 5 minutes ago, now
        observations = await pool_contract.functions.observe(seconds_agos).call()
        tick_cumulatives = observations[0]
        
        avg_tick = (tick_cumulatives[1] - tick_cumulatives[0]) // 300
        twap_price = self._tick_to_price(avg_tick)
        
        # Calculate deviation
        if current_price > twap_price:
            deviation = ((current_price - twap_price) * 10000) // twap_price
        else:
            deviation = ((twap_price - current_price) * 10000) // twap_price
        
        return deviation
    
    except Exception as e:
        # Fallback to conservative estimate
        print(f"Volatility calculation failed: {e}")
        return 300  # 3% default

async def _estimate_pool_liquidity(self, pool_address: str) -> int:
    """
    Estimate pool liquidity in KAS value
    """
    try:
        pool_contract = self.w3.eth.contract(
            address=pool_address,
            abi=UNISWAP_V3_POOL_ABI
        )
        
        token0 = await pool_contract.functions.token0().call()
        token1 = await pool_contract.functions.token1().call()
        
        # Get balances
        token0_contract = self.w3.eth.contract(address=token0, abi=ERC20_ABI)
        token1_contract = self.w3.eth.contract(address=token1, abi=ERC20_ABI)
        
        token0_balance = await token0_contract.functions.balanceOf(pool_address).call()
        token1_balance = await token1_contract.functions.balanceOf(pool_address).call()
        
        # Get current price
        slot0 = await pool_contract.functions.slot0().call()
        sqrt_price_x96 = slot0[0]
        price = self._sqrt_price_to_price(sqrt_price_x96)
        
        # Convert to KAS value
        if token0.lower() == self.WKAS.lower():
            # Token0 is WKAS
            token1_value_in_kas = (token1_balance * price) // 10**18
            total_liquidity = token0_balance + token1_value_in_kas
        else:
            # Token1 is WKAS
            token0_value_in_kas = (token0_balance * 10**18) // price
            total_liquidity = token1_balance + token0_value_in_kas
        
        return total_liquidity
    
    except Exception as e:
        print(f"Liquidity estimation failed: {e}")
        return Web3.to_wei(10000, 'ether')  # 10K KAS fallback

async def _build_swap_transaction(
    self,
    token_in: str,
    token_out: str,
    amount_in: int,
    min_amount_out: int,
    recipient: str
) -> dict:
    """
    Build swap transaction for SwapRouter.exactInputSingle()
    """
    
    swap_router = self.w3.eth.contract(
        address=self.SWAP_ROUTER,
        abi=SWAP_ROUTER_ABI
    )
    
    params = {
        'tokenIn': token_in,
        'tokenOut': token_out,
        'fee': 2500,
        'recipient': recipient,
        'deadline': int(time.time()) + 300,  # 5 minutes
        'amountIn': amount_in,
        'amountOutMinimum': min_amount_out,
        'sqrtPriceLimitX96': 0
    }
    
    # Build transaction
    tx = await swap_router.functions.exactInputSingle(params).build_transaction({
        'from': recipient,
        'value': amount_in if token_in.lower() == self.WKAS.lower() else 0,
        'gas': 300000,  # Estimate
        'maxFeePerGas': await self.w3.eth.max_priority_fee + await self.w3.eth.gas_price,
        'maxPriorityFeePerGas': await self.w3.eth.max_priority_fee,
        'nonce': await self.w3.eth.get_transaction_count(recipient)
    })
    
    return tx

def _sqrt_price_to_price(self, sqrt_price_x96: int) -> int:
    """Convert Uniswap V3 sqrtPriceX96 to human-readable price"""
    return (sqrt_price_x96 ** 2 * 10**18) >> (96 * 2)

def _tick_to_price(self, tick: int) -> int:
    """Convert tick to price"""
    return int(1.0001 ** tick * 10**18)
