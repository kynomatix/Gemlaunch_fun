## Part 2: Implementing GraphQL Client

### Install GraphQL Client

```bash
npm install @apollo/client graphql
# or
yarn add @apollo/client graphql
```

### Setup Apollo Client

```javascript
// lib/apolloClient.js
import { ApolloClient, InMemoryCache, HttpLink, split } from '@apollo/client'
import { GraphQLWsLink } from '@apollo/client/link/subscriptions'
import { getMainDefinition } from '@apollo/client/utilities'
import { createClient } from 'graphql-ws'

const httpLink = new HttpLink({
  uri: 'https://explorer.testnet.kasplextest.xyz/graphiql',
})

const wsLink = new GraphQLWsLink(
  createClient({
    url: 'wss://explorer.testnet.kasplextest.xyz/socket/websocket',
  })
)

// Split based on operation type
const splitLink = split(
  ({ query }) => {
    const definition = getMainDefinition(query)
    return (
      definition.kind === 'OperationDefinition' &&
      definition.operation === 'subscription'
    )
  },
  wsLink,
  httpLink
)

export const apolloClient = new ApolloClient({
  link: splitLink,
  cache: new InMemoryCache(),
})
```

### After (GraphQL Approach)

```javascript
// ✅ Query Blockscout directly
import { gql } from '@apollo/client'
import { apolloClient } from './apolloClient'

const GET_TOKEN_TRANSACTIONS = gql`
  query GetTokenTransactions($address: AddressHash!, $first: Int!) {
    address(hash: $address) {
      transactions(first: $first, order: DESC) {
        edges {
          node {
            hash
            blockNumber
            fromAddressHash
            toAddressHash
            value
            gasUsed
            timestamp
            status
            input
          }
        }
      }
    }
  }
`

async function getRecentTrades(tokenAddress) {
  const { data } = await apolloClient.query({
    query: GET_TOKEN_TRANSACTIONS,
    variables: {
      address: tokenAddress,
      first: 20
    }
  })
  
  return data.address.transactions.edges.map(edge => edge.node)
}
```

## Part 4: React Integration

### Query Component (Fetch Once)

```jsx
import { useQuery, gql } from '@apollo/client'

const GET_TOKEN_INFO = gql`
  query GetTokenInfo($address: AddressHash!) {
    address(hash: $address) {
      hash
      balance
      transactionCount
      tokenTransfers(first: 100) {
        edges {
          node {
            amount
            fromAddressHash
            toAddressHash
            timestamp
          }
        }
      }
    }
  }
`

function TokenInfo({ tokenAddress }) {
  const { loading, error, data } = useQuery(GET_TOKEN_INFO, {
    variables: { address: tokenAddress },
    pollInterval: 10000, // Refetch every 10 seconds
  })
  
  if (loading) return <div>Loading token data...</div>
  if (error) return <div>Error: {error.message}</div>
  
  const transfers = data.address.tokenTransfers.edges.map(e => e.node)
  
  return (
    <div>
      <h2>Token: {tokenAddress}</h2>
      <p>Balance: {data.address.balance}</p>
      <p>Total Transactions: {data.address.transactionCount}</p>
      
      <h3>Recent Transfers</h3>
      <ul>
        {transfers.map((transfer, i) => (
          <li key={i}>
            {transfer.fromAddressHash} → {transfer.toAddressHash}
            <br />
            Amount: {transfer.amount}
            <br />
            Time: {new Date(transfer.timestamp).toLocaleString()}
          </li>
        ))}
      </ul>
    </div>
  )
}
```

### Subscription Component (Real-Time)

```jsx
import { useSubscription, gql } from '@apollo/client'

const NEW_TRANSACTIONS_SUBSCRIPTION = gql`
  subscription OnNewTransaction($address: AddressHash!) {
    addressReceived(addressHash: $address) {
      hash
      value
      fromAddressHash
      toAddressHash
      blockNumber
      timestamp
    }
  }
`

function RecentTrades({ tokenAddress }) {
  const [trades, setTrades] = useState([])
  
  const { data } = useSubscription(NEW_TRANSACTIONS_SUBSCRIPTION, {
    variables: { address: tokenAddress },
    onData: ({ data }) => {
      // Add new transaction to list
      setTrades(prev => [data.data.addressReceived, ...prev].slice(0, 20))
    }
  })
  
  return (
    <div className="recent-trades">
      <h3>Recent Trades (Real-Time)</h3>
      {trades.map(trade => (
        <div key={trade.hash} className="trade-item">
          <span className="address">{trade.fromAddressHash.slice(0, 10)}...</span>
          <span className="arrow">→</span>
          <span className="value">{trade.value} KAS</span>
          <span className="time">{new Date(trade.timestamp).toLocaleTimeString()}</span>
        </div>
      ))}
    </div>
  )
}
```

## Part 7: Example Queries for Your Platform

### Get Token Trade History

```javascript
const GET_TRADE_HISTORY = gql`
  query GetTradeHistory($tokenAddress: AddressHash!, $first: Int!) {
    address(hash: $tokenAddress) {
      tokenTransfers(first: $first, order: DESC) {
        edges {
          node {
            fromAddressHash
            toAddressHash
            amount
            timestamp
            transactionHash
            blockNumber
          }
        }
      }
    }
  }
`

async function getTradeHistory(tokenAddress, limit = 100) {
  const { data } = await apolloClient.query({
    query: GET_TRADE_HISTORY,
    variables: { tokenAddress, first: limit }
  })
  
  const trades = data.address.tokenTransfers.edges.map(({ node }) => ({
    buyer: node.toAddressHash,
    seller: node.fromAddressHash,
    amount: node.amount,
    timestamp: node.timestamp,
    txHash: node.transactionHash,
  }))
  
  return trades
}
```

### Calculate Market Cap from On-Chain Data

```javascript
async function getTokenMarketCap(tokenAddress) {
  // Get contract state
  const { data } = await apolloClient.query({
    query: gql`
      query GetContractState($address: AddressHash!) {
        address(hash: $address) {
          balance
          tokenTransfers(first: 1000) {
            edges {
              node {
                amount
              }
            }
          }
        }
      }
    `,
    variables: { address: tokenAddress }
  })
  
  // Calculate total volume traded
  const totalVolume = data.address.tokenTransfers.edges
    .reduce((sum, { node }) => sum + parseFloat(node.amount), 0)
  
  // Get current KAS price
  const kasPrice = await getKasPrice()
  
  // Calculate using bonding curve
  const bondingCurve = new BondingCurve()
  const { usd: marketCap } = bondingCurve.calculateMarketCap(
    totalVolume,
    kasPrice
  )
  
  return marketCap
}
```

### Get Top Holders

```javascript
const GET_TOP_HOLDERS = gql`
  query GetTopHolders($tokenAddress: AddressHash!) {
    address(hash: $tokenAddress) {
      tokenTransfers {
        edges {
          node {
            toAddressHash
            amount
          }
        }
      }
    }
  }
`

async function getTopHolders(tokenAddress) {
  const { data } = await apolloClient.query({
    query: GET_TOP_HOLDERS,
    variables: { tokenAddress }
  })
  
  // Aggregate by address
  const balances = {}
  
  data.address.tokenTransfers.edges.forEach(({ node }) => {
    const holder = node.toAddressHash
    balances[holder] = (balances[holder] || 0) + parseFloat(node.amount)
  })
  
  // Sort and return top 10
  return Object.entries(balances)
    .sort(([, a], [, b]) => b - a)
    .slice(0, 10)
    .map(([address, balance]) => ({ address, balance }))
}
```

### Subscribe to New Trades for Chart

```javascript
function TradingChart({ tokenAddress }) {
  const [chartData, setChartData] = useState([])
  
  // Load historical data
  useEffect(() => {
    async function loadHistory() {
      const trades = await getTradeHistory(tokenAddress, 1000)
      const candles = convertToOHLC(trades)
      setChartData(candles)
    }
    loadHistory()
  }, [tokenAddress])
  
  // Subscribe to new trades
  useSubscription(NEW_TRANSACTIONS_SUBSCRIPTION, {
    variables: { address: tokenAddress },
    onData: ({ data }) => {
      const newTrade = data.data.addressReceived
      
      // Update chart with new candle
      setChartData(prev => {
        const updated = updateLastCandle(prev, newTrade)
        return updated
      })
    }
  })
  
  return <LightweightChart data={chartData} />
}
```
