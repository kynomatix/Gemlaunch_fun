🔴 CRITICAL SEVERITY FINDINGS
C-1: Incorrect Constant Product AMM Implementation
Location: BondingCurvePool.sol - quoteBuy() and quoteSell()
Issue:
solidity// CURRENT IMPLEMENTATION:
uint256 k = poolTokens * poolKas;
uint256 newPoolKas = poolKas + kasIn;
uint256 newPoolTokens = k / newPoolKas;
tokensOut = poolTokens - newPoolTokens;
Problems:

No fee integration in AMM math - Fees are deducted BEFORE the AMM calculation, which breaks the invariant:

solidity   // In buyTokens():
   uint256 tradeAmount = msg.value - totalFees;  // Fee removed first
   uint256 tokens = quoteBuy(tradeAmount);        // AMM uses post-fee amount
This means the pool balance increases by tradeAmount but the user paid msg.value. The fee amount (totalFees) never enters the pool reserves, creating an accounting gap.

Asymmetric reserve updates - The code shows:

solidity   netReservesBase += tradeAmount;  // Only trade amount added
But address(this).balance includes the full msg.value. The AMM pricing uses address(this).balance in quoteBuy():
solidity   uint256 poolKas = address(this).balance;  // Includes fees!
This creates price drift as fees accumulate.

Integer division precision loss - The formula k / newPoolKas loses precision on every trade, favoring the contract slightly but creating MEV opportunities.

Exploit Scenario:
1. User buys with 100 KAS
2. totalFees = 1.5 KAS deducted
3. tradeAmount = 98.5 KAS goes to AMM
4. But address(this).balance = 100 KAS (includes fees)
5. Next trade prices using 100 KAS, not 98.5 KAS
6. Price is artificially low → arbitrage opportunity
Recommendation:
solidity// Option 1: Fees AFTER AMM calculation (Uniswap v2 pattern)
function buyTokens(uint256 minTokensOut, uint256 deadline) external payable {
    uint256 tokensOutGross = quoteBuy(msg.value);
    uint256 feeInTokens = tokensOutGross * 150 / 10000;
    uint256 tokensOutNet = tokensOutGross - feeInTokens;
    require(tokensOutNet >= minTokensOut, "Slippage");
    // Transfer net tokens to user, keep fee tokens
}

// Option 2: Virtual reserves (cleaner)
uint256 public virtualKasReserve;
uint256 public virtualTokenReserve;

function quoteBuy(uint256 kasIn) public view returns (uint256) {
    uint256 k = virtualTokenReserve * virtualKasReserve;
    uint256 newKasReserve = virtualKasReserve + kasIn;
    return virtualTokenReserve - (k / newKasReserve);
}
Risk: CRITICAL - Enables arbitrage, breaks price discovery

C-2: Fee Accounting Creates Exploitable State Divergence
Location: BondingCurvePool.sol - Fee handling
Issue:
The dual tracking system (grossInBase vs netReservesBase) doesn't align with actual contract balance:
solidity// On buy:

Exploit:

Graduation triggers prematurely (before actual 75 KAS in reserves)
DEX LP is created with insufficient backing
Immediate arbitrage on DEX vs curve price divergence

H-1: Wallet Cap Can Be Bypassed via Flash Loan
Location: BondingCurvePool.sol - _transfer() override
Issue:
solidityfunction _transfer(address from, address to, uint256 amount) internal override {
    if (!graduated) {
        uint256 circulating = totalSupply() - balanceOf(address(this));
        require(
            balanceOf(to) + amount <= (circulating * MAX_WALLET_PCT) / 100,
            "Exceeds 10% wallet cap"
        );
    }
    super._transfer(from, to, amount);
}
Problems:

No check on 'from' address - A whale can:

   1. Hold 10% in wallet A
   2. Transfer 10% to wallet B (passes check on B)
   3. Wallet A now has 0%, can buy another 10%
   4. Wallet B has 10%
   5. Total whale control: 20%

Flash loan bypass:

   1. Flash loan tokens from elsewhere (not this contract)
   2. Transfer to this pool (cap check passes because 'from' is external)
   3. Immediately sell to pool (frontrun other users)
   4. Repay flash loan
   5. Profit from price manipulation

Multi-wallet Sybil - Cap is per-address, not per-entity:

   Attacker creates 10 wallets × 10% each = 100% control
Recommendation:
solidity// Option 1: Cooldown + cap
mapping(address => uint256) public lastTransferTime;

function _transfer(address from, address to, uint256 amount) internal override {
    if (!graduated) {
        require(block.timestamp - lastTransferTime[to] >= 1 hours, "Cooldown");
        
        uint256 circulating = totalSupply() - balanceOf(address(this));
        require(
            balanceOf(to) + amount <= (circulating * MAX_WALLET_PCT) / 100,
            "Exceeds cap"
        );
        
        lastTransferTime[to] = block.timestamp;
    }
    super._transfer(from, to, amount);
}

// Option 2: Global whale tracking (more gas)
mapping(address => bool) public isWhale;
address[] public whales;

function _updateWhaleStatus(address account) internal {
    uint256 circulating = totalSupply() - balanceOf(address(this));
    bool shouldBeWhale = balanceOf(account) > (circulating * 5 / 100); // 5% threshold
    
    if (shouldBeWhale && !isWhale[account]) {
        require(whales.length < 10, "Max whales reached"); // Limit total whales
        isWhale[account] = true;
        whales.push(account);
    }
}
Risk: HIGH - Anti-whale mechanism bypassable

M-3: Treasury Fee Distribution Can Be Blocked
Location: Treasury.sol - distributeFees()
Issue:
solidityfunction _safeTransfer(address to, uint256 amount) private {
    (bool success, ) = payable(to).call{value: amount}("");

Problem: If transfer fails, fees are lost (not reverted to pool):

solidityuint256 public pendingDistribution; // Track failed transfers

function _safeTransfer(address to, uint256 amount) private {
    (bool success, ) = payable(to).call{value: amount}("");
    if (!success) {
        pendingDistribution += amount;  // Track failed amount
        emit TransferFailed(to, amount);
    }
}

function distributeFees() external nonReentrant {
    uint256 balance = address(this).balance - pendingDistribution; // Exclude pending
    require(balance > 0, "No fees to distribute");
    
    // ... normal distribution
}

// Admin can rescue failed transfers
function rescueFailedTransfer(address to, uint256 amount) external onlyAdmin {
    require(amount <= pendingDistribution, "Exceeds pending");
    pendingDistribution -= amount;
    
    (bool success, ) = payable(to).call{value: amount}("");
    require(success, "Rescue failed");
}
