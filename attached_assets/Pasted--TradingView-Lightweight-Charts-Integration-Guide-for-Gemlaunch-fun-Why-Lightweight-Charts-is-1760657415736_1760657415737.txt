## Part 1: Basic Candlestick Chart Setup

### Simple React Implementation

```jsx
import { useEffect, useRef } from 'react'
import { createChart } from 'lightweight-charts'

function TradingChart({ tokenSymbol }) {
  const chartContainerRef = useRef()
  const chartRef = useRef()
  const candlestickSeriesRef = useRef()
  
  useEffect(() => {
    // Create chart
    const chart = createChart(chartContainerRef.current, {
      width: chartContainerRef.current.clientWidth,
      height: 500,
      layout: {
        background: { color: '#1a1a1a' },
        textColor: '#d1d4dc',
      },
      grid: {
        vertLines: { color: '#2B2B43' },
        horzLines: { color: '#363C4E' },
      },
      crosshair: {
        mode: 0, // Normal crosshair
      },
      rightPriceScale: {
        borderColor: '#485c7b',
      },
      timeScale: {
        borderColor: '#485c7b',
        timeVisible: true,
        secondsVisible: false,
      },
    })
    
    // Add candlestick series
    const candlestickSeries = chart.addCandlestickSeries({
      upColor: '#26a69a',
      downColor: '#ef5350',
      borderVisible: false,
      wickUpColor: '#26a69a',
      wickDownColor: '#ef5350',
    })
    
    chartRef.current = chart
    candlestickSeriesRef.current = candlestickSeries
    
    // Handle resize
    const handleResize = () => {
      chart.applyOptions({
        width: chartContainerRef.current.clientWidth,
      })
    }
    window.addEventListener('resize', handleResize)
    
    return () => {
      window.removeEventListener('resize', handleResize)
      chart.remove()
    }
  }, [])
  
  return (
    <div 
      ref={chartContainerRef} 
      style={{ position: 'relative', width: '100%' }}
    />
  )
}

export default TradingChart
```

## Part 2: Loading Data from Kasplex

### Fetch and Format Transaction Data

```javascript
// Helper to fetch Kasplex transactions
async function fetchKasplexTransactions(tokenAddress) {
  try {
    // Replace with your actual Kasplex API endpoint
    const response = await fetch(
      `https://tn10api.kasplex.org/token/${tokenAddress}/transactions`
    )
    const transactions = await response.json()
    
    return transactions
  } catch (error) {
    console.error('Error fetching transactions:', error)
    return []
  }
}

// Convert transactions to OHLC candlestick format
function convertToCandlestickData(transactions, intervalMinutes = 15) {
  const intervalMs = intervalMinutes * 60 * 1000
  const candles = {}
  
  // Sort transactions by timestamp
  transactions.sort((a, b) => a.timestamp - b.timestamp)
  
  transactions.forEach(tx => {
    // Round timestamp to interval
    const candleTime = Math.floor(tx.timestamp / intervalMs) * intervalMs / 1000
    
    const price = calculatePriceFromTx(tx)
    
    if (!candles[candleTime]) {
      // New candle
      candles[candleTime] = {
        time: candleTime,
        open: price,
        high: price,
        low: price,
        close: price,
      }
    } else {
      // Update existing candle
      candles[candleTime].high = Math.max(candles[candleTime].high, price)
      candles[candleTime].low = Math.min(candles[candleTime].low, price)
      candles[candleTime].close = price
    }
  })
  
  // Convert to array and sort
  return Object.values(candles).sort((a, b) => a.time - b.time)
}

// Calculate price from transaction based on bonding curve
function calculatePriceFromTx(tx) {
  // Your bonding curve formula here
  // This depends on your specific implementation
  const tokenAmount = tx.tokenAmount
  const kasAmount = tx.kasAmount
  
  return kasAmount / tokenAmount // Simple example
}
```

### Update Chart with Data

```javascript
// Inside your component
useEffect(() => {
  async function loadData() {
    const transactions = await fetchKasplexTransactions(tokenSymbol)
    const candleData = convertToCandlestickData(transactions, 15) // 15 min candles
    
    if (candlestickSeriesRef.current) {
      candlestickSeriesRef.current.setData(candleData)
    }
  }
  
  loadData()
  
  // Refresh every 30 seconds
  const interval = setInterval(loadData, 30000)
  return () => clearInterval(interval)
}, [tokenSymbol])
```

## Part 3: Fix Bonding Curve Market Cap Calculation

### Proper Bonding Curve Implementation

```javascript
class BondingCurve {
  constructor(config = {}) {
    // Virtual reserves (creates smooth curve)
    this.virtualKasReserves = config.virtualKasReserves || 30
    this.virtualTokenReserves = config.virtualTokenReserves || 1073000000
    
    // Real reserves (actual tokens available)
    this.initialRealTokenReserves = config.initialRealTokenReserves || 793100000
    this.realKasReserves = 0 // Starts at 0
    
    // Total supply
    this.totalSupply = config.totalSupply || 1000000000 // 1 billion
    
    // Graduation threshold
    this.graduationMcap = config.graduationMcap || 69000 // $69k
  }
  
  // Calculate current price based on tokens sold
  getCurrentPrice(tokensSold) {
    const virtualTokensRemaining = this.virtualTokenReserves - tokensSold
    const currentVirtualKas = this.virtualKasReserves
    
    // Price = KAS reserves / Token reserves (constant product)
    return currentVirtualKas / virtualTokensRemaining
  }
  
  // Calculate how much KAS needed to buy X tokens
  calculateKasForTokens(tokenAmount, currentTokensSold) {
    const k = this.virtualKasReserves * this.virtualTokenReserves
    
    const tokensRemaining = this.virtualTokenReserves - currentTokensSold
    const newTokensRemaining = tokensRemaining - tokenAmount
    
    const newKasReserves = k / newTokensRemaining
    const kasNeeded = newKasReserves - this.virtualKasReserves
    
    return kasNeeded
  }
  
  // Calculate real market cap (not fake $1k)
  calculateMarketCap(tokensSold, kasPrice) {
    const pricePerToken = this.getCurrentPrice(tokensSold)
    const marketCapInKas = pricePerToken * this.totalSupply
    const marketCapInUSD = marketCapInKas * kasPrice
    
    return {
      kas: marketCapInKas,
      usd: marketCapInUSD
    }
  }
  
  // Get progress towards graduation
  getBondingCurveProgress(tokensSold) {
    return (tokensSold / this.initialRealTokenReserves) * 100
  }
  
  // Calculate price impact for a trade
  getPriceImpact(tokenAmount, currentTokensSold) {
    const currentPrice = this.getCurrentPrice(currentTokensSold)
    const averagePrice = this.calculateKasForTokens(tokenAmount, currentTokensSold) / tokenAmount
    
    return ((averagePrice - currentPrice) / currentPrice) * 100
  }
}

// Usage example
const bondingCurve = new BondingCurve({
  virtualKasReserves: 30,
  virtualTokenReserves: 1073000000,
  initialRealTokenReserves: 793100000,
  totalSupply: 1000000000,
  graduationMcap: 69000
})

// When processing a transaction
const tokensSold = 50000000 // 50M tokens sold so far
const kasPrice = 0.15 // Current KAS price in USD

const { usd: marketCap } = bondingCurve.calculateMarketCap(tokensSold, kasPrice)
console.log(`Real Market Cap: $${marketCap.toFixed(2)}`)

// This will show the ACTUAL market cap, not a fake $1k
```

### Frontend WebSocket Integration

```javascript
function TradingChart({ tokenAddress }) {
  const chartContainerRef = useRef()
  const chartRef = useRef()
  const candlestickSeriesRef = useRef()
  const wsRef = useRef()
  
  useEffect(() => {
    // ... chart creation code from earlier ...
    
    // Connect to WebSocket
    const ws = new WebSocket(`wss://your-api.com/ws?token=${tokenAddress}`)
    wsRef.current = ws
    
    ws.onmessage = (event) => {
      const message = JSON.parse(event.data)
      
      if (message.type === 'trade') {
        // Update the last candle or create a new one
        updateChartWithTrade(message.data)
      }
    }
    
    ws.onerror = (error) => {
      console.error('WebSocket error:', error)
    }
    
    return () => {
      ws.close()
    }
  }, [tokenAddress])
  
  const updateChartWithTrade = (trade) => {
    if (!candlestickSeriesRef.current) return
    
    // Get current candle data
    const currentTime = Math.floor(Date.now() / 1000)
    const intervalSeconds = 15 * 60 // 15 minutes
    const candleTime = Math.floor(currentTime / intervalSeconds) * intervalSeconds
    
    // Update method depends on if it's a new candle or updating existing
    candlestickSeriesRef.current.update({
      time: candleTime,
      open: trade.price, // You'd track this properly
      high: trade.price,
      low: trade.price,
      close: trade.price,
    })
  }
  
  return <div ref={chartContainerRef} style={{ width: '100%', height: '500px' }} />
}
```

## Part 6: Kasplex Integration Code

### Connect to Kasplex Testnet

```javascript
import { Indexer } from 'KasplexBuilder'

class KasplexDataProvider {
  constructor(network = 'testnet') {
    this.indexer = new Indexer(
      network === 'testnet' 
        ? 'https://tn10api.kasplex.org'
        : 'https://api.kasplex.org'
    )
    this.bondingCurve = new BondingCurve()
  }
  
  async getTokenTransactions(tokenAddress, fromBlock, toBlock) {
    try {
      // Fetch from Kasplex indexer
      const response = await fetch(
        `${this.indexer.apiUrl}/token/${tokenAddress}/transactions?from=${fromBlock}&to=${toBlock}`
      )
      const data = await response.json()
      
      return data.transactions
    } catch (error) {
      console.error('Error fetching transactions:', error)
      return []
    }
  }
  
  async getTokenInfo(tokenAddress) {
    try {
      const response = await fetch(
        `${this.indexer.apiUrl}/token/${tokenAddress}`
      )
      return await response.json()
    } catch (error) {
      console.error('Error fetching token info:', error)
      return null
    }
  }
  
  async getCurrentMarketCap(tokenAddress) {
    const tokenInfo = await this.getTokenInfo(tokenAddress)
    if (!tokenInfo) return 0
    
    const kasPrice = await this.getKasPrice()
    const { usd } = this.bondingCurve.calculateMarketCap(
      tokenInfo.totalSold,
      kasPrice
    )
    
    return usd
  }
  
  async getKasPrice() {
    try {
      // Fetch KAS price from CoinGecko or similar
      const response = await fetch(
        'https://api.coingecko.com/api/v3/simple/price?ids=kaspa&vs_currencies=usd'
      )
      const data = await response.json()
      return data.kaspa.usd
    } catch (error) {
      console.error('Error fetching KAS price:', error)
      return 0.15 // Fallback price
    }
  }
}

  const marketCap = await provider.getCurrentMarketCap(tokenAddress)

}
```
